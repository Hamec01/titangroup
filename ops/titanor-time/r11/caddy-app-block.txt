# R11 — Caddy-блок для app.titanorgroup.fi (holding / maintenance).
# Вставляется в /etc/caddy/Caddyfile ПОСЛЕ блока t97-dd686bc3d4.84.247.130.242.nip.io.
# Применять по docs/titanor-time/R11_DOMAIN_CADDY_PLAN_RU.md §3.3 ПОСЛЕ того,
# как владелец создал DNS-запись A app -> 84.247.130.242 (DNS only) в Cloudflare.
#
# Вариант A (grey-cloud): Caddy сам берёт сертификат Let's Encrypt, TITANOR_TRUSTED_PROXY_HOPS=1.
# Проверено `caddy validate` + `caddy adapt` на Caddy 2.6.2.
#
# Заголовки безопасности продублированы в handle_errors: ответ 503 рендерится ОТДЕЛЬНЫМ
# error-маршрутом, мимо site-level `header` (проверено вживую 2026-08-31 — без дубля
# HSTS/X-Robots/-Server на 503 не попадали).
#
# НА R14 (production cutover): удалить `root`, оба `header {}`, `handle {}`, `handle_errors {}`
# и добавить одну строку:  reverse_proxy 127.0.0.1:3199

app.titanorgroup.fi {
	encode zstd gzip
	root * /var/www/titanor-time-holding

	header {
		X-Content-Type-Options nosniff
		Referrer-Policy strict-origin-when-cross-origin
		X-Frame-Options DENY
		Strict-Transport-Security "max-age=63072000"
		X-Robots-Tag "noindex, nofollow"
		Cache-Control "no-store"
		-X-Powered-By
		-Server
	}

	handle {
		error "Titanor Time not launched yet" 503
	}

	handle_errors {
		header {
			X-Content-Type-Options nosniff
			Referrer-Policy strict-origin-when-cross-origin
			X-Frame-Options DENY
			Strict-Transport-Security "max-age=63072000"
			X-Robots-Tag "noindex, nofollow"
			Cache-Control "no-store"
			-X-Powered-By
			-Server
		}
		rewrite * /index.html
		file_server
	}

	log {
		output file /var/log/caddy/titanor-time-app-access.log
		format json
	}
}
